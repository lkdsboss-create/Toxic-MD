// ================= FIX 4 : STARTTOXIC SANS BOUCLE SPAM =================
async function startToxic() {
  if (global._toxicIsStarting) {
    console.log('⚠️ startToxic déjà en cours, skip');
    return;
  }
  if (global._toxicShuttingDown) return;
  
  global._toxicIsStarting = true;

  try {
    console.log(chalk.cyan('🔄 Initialisation Toxic-MD...'));
    
    // Charge la session depuis ENV ou fichier
    await authenticationn(); // ton auth.js qui décode SESSION_ID
    
    const { state, saveCreds } = await useMultiFileAuthState(sessionName);
    
    // FIX QR REFS ATTEMPTS ENDED : on désactive QR si SESSION fournie
    const hasSession = fs.existsSync(path.join(sessionName, 'creds.json'));
    if (!hasSession) {
      console.log(chalk.red('❌ AUCUNE SESSION TROUVEE !'));
      console.log(chalk.yellow('→ Va sur https://toxicx.tech/pairing et mets SESSION_ID en variable env'));
      console.log(chalk.yellow('→ Sans SESSION, Baileys va essayer de générer un QR qui timeout en 20s (erreur 408)'));
      // On ne quitte pas, on laisse essayer 1 fois mais on prévient
    }

    const logger = pino({ level: 'silent' });
    
    const client = toxicConnect({
      auth: {
        creds: state.creds,
        keys: makeCacheableSignalKeyStore(state.keys, logger),
      },
      logger,
      browser: Browsers.macOS('Safari'),
      markOnlineOnConnect: false, // FIX: évite de supprimer les notifs téléphone
      syncFullHistory: false,
      shouldSyncHistoryMessage: () => false,
      printQRInTerminal: false, // FIX: pas de QR sur serveur cloud
      getMessage: async (key) => {
        const msg = store.loadMessage(key.remoteJid, key.id);
        return msg?.message || undefined;
      }
    });

    currentSock = client;
    global._toxicCurrentClient = client;
    store.bind(client.ev);

    // Cache LID
    const lidPhoneCache = new Map();
    const phoneLidCache = new Map();

    client.ev.on('creds.update', saveCreds);

    client.ev.on('connection.update', async (update) => {
      const { connection, lastDisconnect } = update;
      
      if (connection === 'open') {
        reconnectAttempts = 0;
        console.log(chalk.green('\n╭───( Toxic-MD )───'));
        console.log(chalk.green('> 🚀 Connected Successfully'));
        console.log(chalk.green(`> Status : Started Successfully ✓`));
        console.log(chalk.green(`> Mode : public`));
        console.log(chalk.green('╰───────────────\n'));
      }

      if (connection === 'close') {
        if (global._toxicShuttingDown) return;
        
        const statusCode = new Boom(lastDisconnect?.error)?.output?.statusCode;
        const reason = lastDisconnect?.error?.output?.payload?.error || statusCode;
        
        console.log(chalk.yellow(`🔴 CONNECTION CLOSED - CODE: ${statusCode} - REASON: ${reason}`));

        // FIX BOUCLE 401 : si déconnecté définitivement, on stoppe
        if (statusCode === DisconnectReason.loggedOut || statusCode === 401) {
          global._toxicShuttingDown = true;
          if (global._toxicReconnectTimer) { clearTimeout(global._toxicReconnectTimer); global._toxicReconnectTimer = null; }
          try { fs.rmSync(sessionName, { recursive: true, force: true }); } catch {}
          console.log(chalk.red('\n╭─❏ SESSION LOGGED OUT'));
          console.log(chalk.red('│ Session invalide (401). Régénère une SESSION sur toxicx.tech'));
          console.log(chalk.red('│ Supprime Session/ et redémarre'));
          console.log(chalk.red('╰───────────────\n'));
          setTimeout(() => process.exit(0), 2000);
          return;
        }

        // FIX BOUCLE 408 QR REFS : timeout QR/pairing
        if (statusCode === 408) {
          console.log(chalk.red('❌ QR refs attempts ended (408) -> SESSION manquante ou invalide'));
          console.log(chalk.yellow('→ Mets bien SESSION_ID en variable env, ne compte pas sur le QR'));
          const delay = 15000; // attend 15s avant retry
          if (!global._toxicReconnectTimer) {
            global._toxicReconnectTimer = setTimeout(() => {
              global._toxicReconnectTimer = null;
              global._toxicIsStarting = false;
              startToxic();
            }, delay);
          }
          global._toxicIsStarting = false;
          return;
        }

        // Autres déconnexions temporaires (500, 503, 515, timedOut)
        if ([DisconnectReason.connectionClosed, DisconnectReason.connectionLost, DisconnectReason.timedOut, 500, 503, 515].includes(statusCode)) {
          const delay = Math.min(baseReconnectDelay * Math.pow(1.5, reconnectAttempts), 30000);
          reconnectAttempts++;
          console.log(chalk.yellow(`🔄 Reconnexion temporaire (raison: ${statusCode}) dans ${delay}ms - tentative ${reconnectAttempts}`));
          if (!global._toxicReconnectTimer) {
            global._toxicReconnectTimer = setTimeout(() => {
              global._toxicReconnectTimer = null;
              global._toxicIsStarting = false;
              startToxic();
            }, delay);
          }
          global._toxicIsStarting = false;
          return;
        }

        // Retry normal
        if (reconnectAttempts < maxReconnectAttempts) {
          const delay = Math.min(baseReconnectDelay * Math.pow(1.5, reconnectAttempts), 60000);
          reconnectAttempts++;
          console.log(chalk.yellow(`🔄 Reconnexion dans ${delay}ms (tentative ${reconnectAttempts}/${maxReconnectAttempts})`));
          if (!global._toxicReconnectTimer) {
            global._toxicReconnectTimer = setTimeout(() => {
              global._toxicReconnectTimer = null;
              global._toxicIsStarting = false;
              startToxic();
            }, delay);
          }
        } else {
          console.log(chalk.red('❌ Trop de tentatives, pause 30s'));
          reconnectAttempts = 0;
          if (!global._toxicReconnectTimer) {
            global._toxicReconnectTimer = setTimeout(() => {
              global._toxicReconnectTimer = null;
              global._toxicIsStarting = false;
              startToxic();
            }, 30000);
          }
        }
        global._toxicIsStarting = false;
        return;
      }

      try { await connectionHandler(client, update, startToxic); } catch {}
    });

    // Handlers messages
    client.ev.on('messages.upsert', async (m) => {
      try {
        const msg = m.messages[0];
        if (!msg.message) return;
        // ton code smsg etc...
        await smsg(client, msg, store);
      } catch (e) {
        console.error('messages.upsert error', e);
      }
    });

    client.sendText = (jid, text, quoted = "", options) => client.sendMessage(jid, { text, ...options }, { quoted });

    client.downloadMediaMessage = async (message) => {
      let mime = (message.msg || message).mimetype || '';
      let messageType = message.mtype ? message.mtype.replace(/Message/gi, '') : mime.split('/')[0];
      const stream = await downloadContentFromMessage(message, messageType);
      let buffer = Buffer.from([]);
      for await (const chunk of stream) buffer = Buffer.concat([buffer, chunk]);
      return buffer;
    };

    global._toxicIsStarting = false;

  } catch (error) {
    console.log('❌ [START TOXIC ERROR]:', error);
    global._toxicCurrentClient = null;
    global._toxicIsStarting = false;
    if (!global._toxicReconnectTimer) {
      global._toxicReconnectTimer = setTimeout(() => {
        global._toxicReconnectTimer = null;
        startToxic();
      }, 5000);
    }
  }
}

startBackupInterval();
startToxic();
